package jrp.progra.proyecto_1;

import java.io.*;
import java.net.*;
import java.util.function.Consumer;

/**
 * Clase encargada de la comunicación entre dos computadoras.
 * Utiliza un hilo secundario (Thread) para evitar que la interfaz de JavaFX se congele.
 */
public class Conexion {
    private Socket socket;
    private PrintWriter out;
    private BufferedReader in;
    private boolean conectado = false;

    // Este "callback" sirve para avisar al controlador cuando llega un puntaje del oponente
    private Consumer<Integer> onPuntajeRecibido;

    /**
     * Modo Servidor: Abre un puerto y espera a que el otro jugador se conecte.
     */
    new Thread(() -> {
            try {
                ServerSocket serverSocket = new ServerSocket(puerto);
                System.out.println("Esperando al oponente...");
                this.soc = serverSocket.accept(); 
                System.out.println("Oponente conectado!");
                iniciar();
            } catch (IOException e) {
                System.err.println("Error en servidor: " + e.getMessage());
            }
        }).start();

    /**
     * Modo Cliente: Se conecta a la dirección IP de la otra computadora.
     */
    public void iCliente(String ip, int puerto) {
        // También ponemos el cliente en un Thread por si la IP tarda en responder
        new Thread(() -> {
            try {
                this.soc = new Socket(ip, puerto);
                System.out.println("Conectado al servidor!");
                iniciar();
            } catch (IOException e) {
                System.err.println("Error al conectar: " + e.getMessage());
            }
        }).start();
    }

    private void configurarFlujos() throws IOException {
        // PrintWriter para enviar datos y BufferedReader para recibirlos
        this.out = new PrintWriter(socket.getOutputStream(), true);
        this.in = new BufferedReader(new InputStreamReader(socket.getInputStream()));
        this.conectado = true;
        
        // INICIO DEL THREAD: Creamos un hilo separado para escuchar la red
        // Esto es vital para que la ventana del juego no se quede "congelada"
        new Thread(() -> {
            try {
                String mensaje;
                // Mientras estemos conectados, el hilo se queda esperando un mensaje
                while (conectado && (mensaje = in.readLine()) != null) {
                    try {
                        int puntosOponente = Integer.parseInt(mensaje);
                        // Avisamos al controlador que el puntaje cambió
                        if (onPuntajeRecibido != null) {
                            onPuntajeRecibido.accept(puntosOponente);
                        }
                    } catch (NumberFormatException e) {
                        System.out.println("Mensaje no válido recibido: " + mensaje);
                    }
                }
            } catch (IOException e) {
                System.err.println("Conexión perdida: " + e.getMessage());
                conectado = false;
            }
        }).start(); // .start() es lo que pone en marcha el hilo secundario
    }

    /**
     * Envía el puntaje actual al otro jugador a través de la red.
     */
    public void enviarPuntaje(int puntos) {
        if (conectado && out != null) {
            out.println(puntos); // Envía el número como texto seguido de un salto de línea
        }
    }

    /**
     * Permite al controlador definir qué hacer cuando llegue un puntaje del oponente.
     */
    public void setOnPuntajeRecibido(Consumer<Integer> callback) {
        this.onPuntajeRecibido = callback;
    }

    public void cerrarConexion() {
        try {
            conectado = false;
            if (socket != null) socket.close();
        } catch (IOException e) {
            e.printStackTrace();
        }
    }

    public boolean isConectado() {
        return conectado;
    }
}
